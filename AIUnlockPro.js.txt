const UA = "Mozilla/5.0 (iPhone; CPU iPhone OS 18_0 like Mac OS X) AppleWebKit/605.1.15 Version/18.0 Mobile/15E148 Safari/604.1";

const services = [
  { name: "ChatGPT", url: "https://chatgpt.com/" },
  { name: "OpenAI", url: "https://api.openai.com/v1/models" },
  { name: "Claude", url: "https://claude.ai/" },

  { name: "Character.AI", url: "https://character.ai/" }
];

const policy = $environment.params || "当前节点";
const opts = { policy: $environment.params };

function escapeHTML(s) {
  return String(s || "").replace(/[&<>"']/g, c => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;",
    '"': "&quot;", "'": "&#39;"
  })[c]);
}

function fetchData(url, timeout) {
  return $task.fetch({
    url: url,
    method: "GET",
    opts: opts,
    timeout: timeout || 9000,
    headers: {
      "User-Agent": UA,
      "Accept": "text/html,application/json,*/*"
    }
  });
}

function checkService(item) {
  const start = Date.now();

  return fetchData(item.url, 9000).then(response => {
    const elapsed = Date.now() - start;
    const code = response.statusCode;
    const body = String(response.body || "").toLowerCase();

    const regionBlocked =
      /unsupported country|not available in your country|not available in your region|country is not supported|region is not supported|service is unavailable in your region|location is not supported/.test(body);

    let status = "未知";
    let icon = "❔";

    if (regionBlocked || code === 451) {
      status = "地区受限";
      icon = "🚫";
    } else if (code === 403) {
      status = "访问被拒";
      icon = "⚠️";
    } else if (code === 429) {
      status = "请求频率受限";
      icon = "⚠️";
    } else if (code >= 200 && code < 400) {
      status = "网站可访问";
      icon = "✅";
    } else if (code >= 500) {
      status = "平台服务器异常";
      icon = "⚠️";
    } else {
      status = "HTTP异常";
      icon = "❔";
    }

    return {
      name: item.name,
      status: status,
      icon: icon,
      code: code,
      time: elapsed
    };
  }).catch(error => {
    return {
      name: item.name,
      status: "连接失败",
      icon: "❌",
      code: "TIMEOUT/ERROR",
      time: Date.now() - start
    };
  });
}

function getIPInfo() {
  return fetchData("https://ipapi.co/json/", 9000)
    .then(response => {
      const data = JSON.parse(response.body || "{}");
      if (response.statusCode !== 200 || !data.ip) {
        throw new Error("IP查询失败");
      }
      return {
        ip: data.ip,
        country: data.country_name || "未知",
        code: data.country_code || "--",
        city: data.city || "未知",
        org: data.org || "未知"
      };
    })
    .catch(() => ({
      ip: "获取失败",
      country: "未知",
      code: "--",
      city: "未知",
      org: "未知"
    }));
}
